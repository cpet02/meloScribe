# meloScribe dependencies
#
# Install torch FIRST, from the PyTorch index, or pip will pull the CPU-only
# build and the GPU sits idle. See README "Setup" for the exact command.

# --- core -------------------------------------------------------------------
numpy
scipy                 # bandpass filtering in audio.py
librosa               # CQT, PYIN, chroma, onsets, resampling
soundfile             # WAV I/O
demucs                # source separation (used via its Python API, not the CLI)
basic-pitch           # note/onset posteriorgram voter

# --- lyrics -----------------------------------------------------------------
mutagen               # ID3 tags, to prefill the track name
# LRClib is reached with urllib from the standard library - no client library.

# --- web UI -----------------------------------------------------------------
fastapi
uvicorn[standard]
python-multipart      # required by FastAPI for file uploads
pydantic

# --- output -----------------------------------------------------------------
pretty_midi           # MIDI export

# --- evaluation -------------------------------------------------------------
mir_eval              # melody and transcription metrics
pytest

# --- optional, but each one measurably improves results ---------------------
# torchcrepe          # the strongest pitch voter; needs torch
# faster-whisper      # transcribe lyrics when LRClib has none
# torchaudio          # word-level forced alignment (install with torch)
